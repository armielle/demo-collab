\section*{\color{bleu}3. Contenu et objectifs}

\footnotesize
\begin{enumerate}[itemsep=3pt]
  \item \textbf{Qu'est-ce qu'apprendre ?} Apprentissage supervisé, non supervisé, par
        renforcement ; le pipeline données $\rightarrow$ modèle $\rightarrow$ décision ;
        risque empirique et généralisation ; compromis biais-variance ; découpage
        train/validation/test et validation croisée ; \textbf{fuite de données}.
        \textit{Objectif : savoir ce qu'un score mesure, et ce qu'il ne mesure pas.}

  \item \textbf{Préparer les données.} Valeurs manquantes (suppression, imputation
        moyenne/médiane/KNN) ; encodage one-hot, ordinal, cible ; variables cycliques ;
        standardisation et outliers. \textbf{Le prétraitement vit dans le pipeline},
        jamais avant le split. \textit{Objectif : ne jamais laisser fuiter le test.}

  \item \textbf{Régression et régularisation.} Moindres carrés (forme fermée, hypothèses,
        lecture des coefficients) ; descente de gradient et pas d'apprentissage ;
        Ridge, Lasso, ElasticNet ; résidus, R², RMSE/MAE, multicolinéarité.
        \textit{Objectif : choisir et diagnostiquer un modèle linéaire.}

  \item \textbf{Classification linéaire et évaluation.} Régression logistique, sigmoïde,
        entropie croisée ; \textbf{le seuil est un choix métier}, pas 0,5 par défaut ;
        matrice de confusion, précision, rappel, F1, ROC/AUC ; classes déséquilibrées.
        \textit{Objectif : justifier une métrique par le coût des erreurs.}

  \item \textbf{Arbres, ensembles et interprétabilité.} CART, Gini et entropie, élagage ;
        bagging, forêt aléatoire ; gradient boosting ; importance des variables,
        dépendance partielle, SHAP. \textit{Objectif : expliquer une décision, pas seulement
        la prédire.}

  \item \textbf{SVM et noyaux.} Marge maximale, vecteurs de support, marge souple (C) ;
        astuce du noyau, RBF et effet de $\gamma$ ; quand le SVM bat les arbres.
        \textit{Objectif : reconnaître les situations où il est le bon outil.}

  \item \textbf{Apprentissage non supervisé.} k-means (initialisation, coude, silhouette) ;
        CAH et DBSCAN ; ACP, variance expliquée, t-SNE/UMAP ; détection d'anomalies
        (Isolation Forest). \textit{Objectif : interpréter une structure sans étiquette.}

  \item \textbf{Fiabilité, éthique et production.} Surapprentissage par recherche
        d'hyperparamètres ; biais, équité, variables sensibles, protection des données ;
        dérive des données, surveillance, versionnage ; arbitrage interprétable/performant.
        \textit{Objectif : livrer un modèle dont on assume les limites.}
\end{enumerate}
\normalsize
