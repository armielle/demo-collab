\section*{\color{bleu}1. Objectifs}
\begin{obj}{À la fin de ce TP, vous saurez}
\begin{itemize}
  \item charger un jeu de données réel et le découper en entraînement et test ;
  \item construire une \textbf{baseline} et interpréter un score par rapport à elle ;
  \item entraîner un modèle dans un \textbf{pipeline} scikit-learn ;
  \item lire une matrice de confusion et une validation croisée ;
  \item \textbf{fabriquer une fuite de données}, puis la corriger.
\end{itemize}
\end{obj}

\section*{\color{bleu}2. Prérequis et matériel}
{\small
Python (variables, boucles, fonctions), notion de moyenne et d'écart-type.
Notebook \texttt{TP1\_premier\_pipeline.ipynb}, noyau \texttt{ml-60h}.
Aucun téléchargement : le jeu est fourni par scikit-learn.}

\section*{\color{bleu}3. Données}
{\small
Jeu \textbf{breast\_cancer} : 569 patients, 30 mesures, étiquette bénin / malin.
C'est un jeu de référence, déjà nettoyé : l'objectif du TP n'est pas la
préparation (chapitre 2) mais la \textbf{démarche d'évaluation}.}

\section*{\color{bleu}7. Règles}

\begin{attn}{Ce qui coûte des points}
\begin{itemize}
  \item Un chiffre annoncé sans la cellule qui le produit ne rapporte rien.
  \item Un notebook qui ne s'exécute pas de bout en bout est plafonné à 5/10.
  \item Une réponse du type « c'est mieux » sans chiffre n'est pas une réponse.
  \item Graine fixée : deux exécutions doivent donner le même résultat.
\end{itemize}
\end{attn}
