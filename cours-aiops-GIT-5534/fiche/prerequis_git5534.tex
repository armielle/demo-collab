\begin{pbox}{1. Prérequis indispensables}
\begin{description}[style=nextline]
\item[Programmation Python (syntaxe de base)] variables, fonctions, boucles,
listes et dictionnaires, \texttt{import}, lecture/écriture de fichier,
lire un message d'erreur (\emph{traceback}) sans paniquer. \emph{Remarque :
NumPy et Pandas sont enseignés en séance 1 --- mais on ne peut pas les apprendre
sans la syntaxe de base.}
\item[Mathématiques de niveau bac] pourcentages et proportions (calculer une
disponibilité de 99,9 \%), moyenne, variance, lire une échelle logarithmique,
manipuler une formule simple ($z$, IQR, $F_1$).
\item[Statistiques élémentaires (bac/1\textsuperscript{re} année)] moyenne,
médiane, distribution, corrélation --- le vocabulaire doit être familier ;
il sera formalisé en séance 2.
\item[Notions de systèmes] serveur, CPU, RAM, disque, salle serveur ;
différence application/système.
\item[Notions de réseaux] adresse IP, port, pare-feu, latence versus débit,
ce qu'est une authentification (login/mot de passe).
\end{description}
\end{pbox}

\vspace{2mm}
\begin{abox}{2. À faire AVANT la première séance (30--45 min)}
\begin{enumerate}
\item Installer \textbf{Anaconda} (Python 3.11) sur votre ordinateur :
\href{https://www.anaconda.com/download}{anaconda.com/download} ;
\item vérifier que vous savez ouvrir un \textbf{notebook Jupyter} et exécuter une cellule ;
\item vérifier les bibliothèques en tapant dans un notebook :\\
\texttt{import numpy, pandas, matplotlib, seaborn, sklearn} --- si une ligne échoue :
\texttt{pip install <nom>} dans Anaconda Prompt ;
\item créer un dossier de cours et y placer le dossier \texttt{data/} distribué
(4 fichiers CSV : métriques propres, métriques brutes, logs, dataset supervisé) ;
\item apporter \textbf{votre portable} (tout le cours tourne sur CPU : un portable
ordinaire suffit, aucun GPU n'est nécessaire).
\end{enumerate}
\end{abox}

\begin{rbox}{3. Ce qui n'est PAS un prérequis --- ne vous découragez pas}
\begin{itemize}
\item \textbf{Aucune expérience en machine learning} : les trois familles de ML
sont enseignées dès la séance 1, à partir de zéro.
\item \textbf{Aucune statistique avancée} : pas de tests d'hypothèses formels
avant la séance 2, qui reprend tout.
\item \textbf{Aucune expérience DevOps} : le vocabulaire d'exploitation
(SLA, SLO, MTBF) est défini en séance 1.
\item \textbf{Pas de machine puissante} : graines fixées, calculs allégés,
un portable standard suffit pour l'intégralité des TP.
\end{itemize}
\end{rbox}
